public class Solution {
    public static void alphaHill(int n) {

        for(int i = 0; i < n; i++) {
            for(int s = 0; s < n - i - 1; s++) {
                System.out.print("  "); 
            }
            for(int j = 0; j <= 2 * i; j++) {

                if(j <= i) {
                    System.out.print((char)('A' + j) + " ");
                } else {
                    System.out.print((char)('A' + (2 * i - j)) + " ");
                }

            }

            System.out.println();
        }
 
    }
}
