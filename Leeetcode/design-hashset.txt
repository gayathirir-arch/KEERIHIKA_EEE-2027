class MyHashSet {
    boolean[] MyHash;
    public MyHashSet() {
        MyHash=new boolean[1000001];
    }
    
    public void add(int key) {
        MyHash[key]=true;

    }
    
    public void remove(int key) {
        MyHash[key]=false;
    }
    public boolean contains(int key) {
        return MyHash[key];
    }
}
